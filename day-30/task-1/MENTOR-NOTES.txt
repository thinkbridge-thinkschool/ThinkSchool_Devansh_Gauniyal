Transactional outbox — writes to the SAME DbContext instance the approving request's
own repository call uses, so the outbox row and the invoice's state change land in one
transaction without this class managing that guarantee itself:

public sealed class EfIntegrationEventOutbox(InvoicingDbContext dbContext) : IIntegrationEventOutbox
{
    public Task EnqueueInvoiceApprovedAsync(InvoiceApproved domainEvent, CancellationToken cancellationToken)
    {
        var payload = new InvoiceApprovedIntegrationEvent(
            domainEvent.InvoiceId.Value,
            domainEvent.ApprovalKind.ToString(),
            domainEvent.ApprovedBy,
            domainEvent.ConsumedAmount.Amount,
            domainEvent.ConsumedAmount.Currency,
            domainEvent.OccurredAt);

        dbContext.OutboxMessages.Add(new OutboxMessage
        {
            Id = Guid.NewGuid(),
            EventType = nameof(InvoiceApprovedIntegrationEvent),
            Payload = JsonSerializer.Serialize(payload),
            OccurredAt = domainEvent.OccurredAt
        });

        return Task.CompletedTask;
    }
}

Best-effort supplier notification — the opposite pattern by design: every send is
wrapped so a failure is logged and swallowed, never rethrown, since a notification must
never fail the request it's attached to:

public async Task NotifyInvoiceApprovedAsync(Guid supplierId, string invoiceNumber, CancellationToken cancellationToken)
{
    try
    {
        var sender = client.CreateSender(topicName);
        var message = new ServiceBusMessage(JsonSerializer.Serialize(new { supplierId, invoiceNumber, kind = "Approved" }));
        await sender.SendMessageAsync(message, cancellationToken);
    }
    catch (Exception ex)
    {
        logger.LogWarning(ex, "Failed to notify supplier {SupplierId} of invoice {InvoiceNumber} approval - swallowed, not fatal to the request.", supplierId, invoiceNumber);
    }
}

Deemed-approval sweep as a real hosted service, running against the deployed clock, in
a fresh DI scope per tick (a live 10-day review window can't be waited out in a working
session, so POST /v1/system/deemed-approval-sweep also exposes the same use case
on demand for live proof against real Azure SQL):

public sealed class DeemedApprovalSweepBackgroundService(
    IServiceScopeFactory scopeFactory, IConfiguration configuration,
    ILogger<DeemedApprovalSweepBackgroundService> logger) : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        var minutes = configuration.GetValue("DeemedApprovalSweep:IntervalMinutes", 15);
        using var timer = new PeriodicTimer(TimeSpan.FromMinutes(minutes));

        while (await timer.WaitForNextTickAsync(stoppingToken))
        {
            await using var scope = scopeFactory.CreateAsyncScope();
            var useCase = scope.ServiceProvider.GetRequiredService<ApplyDeemedApprovalsUseCase>();
            var unitOfWork = scope.ServiceProvider.GetRequiredService<IUnitOfWork>();

            var deemed = await useCase.ExecuteAsync(stoppingToken);
            if (deemed.Count > 0)
            {
                await unitOfWork.SaveChangesAsync(stoppingToken);
                logger.LogInformation("Deemed-approval sweep approved {Count} invoice(s).", deemed.Count);
            }
        }
    }
}
